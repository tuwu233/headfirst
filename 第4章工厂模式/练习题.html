    def order(self):
        return self.create() + "->bake"

    def create(self):
        return "base"

    def create(self):
        return "NY"


print(NYStore().order())
QUESTION 08单选题10 分创建与复用
以下代码两次通过同一个工厂创建产品。打印什么？
calls = []


def create():
    calls.append("created")
    return []


def use(factory):
    return factory()


a = use(create)
b = use(create)
print(a is b, len(calls))

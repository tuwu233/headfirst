QUESTION 07单选题10 分初始化陷阱
这是故意包含重复初始化问题的教学代码。输出是什么？
class Counter:
    _instance = None

    def __new__(cls):
        if cls._instance is None:
            cls._instance = super().__new__(cls)
        return cls._instance

    def __init__(self):
        self.value = 0


a = Counter()
a.value = 7
b = Counter()
print(a is b, a.value)
